{
  "session_id": "your-session",
  "current_tokens": 54000,
  "strategy": "balanced"
}
// response
→ { "should_prune": true, "tokens_to_prune": 39000, "estimated_cost_saved_cents": 8.42 }
POST /api/v1/context/analyze 
POST /api/v1/context/analyze
{
  "session_id": "your-session",
  "token_history": [4000, 8200, 13100, 19500, 27800],
  "strategy": "adaptive"
}
// response
→ { "recommended_threshold": 26410, "growth_rate_per_msg": 5950, "strategy": "adaptive" }
